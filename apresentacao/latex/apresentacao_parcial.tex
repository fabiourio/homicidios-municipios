\begin{frame}{1 · Questão de pesquisa}
\textcolor{navy}{\textbf{Quais características socioeconômicas, demográficas e geográficas estão associadas à taxa de homicídios nos municípios brasileiros?}}\par
\vspace{1em}
\begin{itemize}
\item \nMunBR{} municípios
\item \nHomBR{} homicídios em 2021--2023
\item \alert{\nTaxaBR} homicídios por 100 mil hab. por ano
\end{itemize}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Limitações dos dados}
\begin{itemize}
\setlength{\itemsep}{0.5em}
\item Estudo observacional: associação, não causa
\item Só homicídios registrados: há subnotificação
\item Presos contam como moradores do município do presídio
\item Municípios pequenos: poucos casos e muitos zeros, a taxa oscila por acaso
\end{itemize}
\end{frame}

\begin{frame}{3 · Variáveis}

\begin{columns}[T]
\column{0.6\textwidth}
\textbf{\color{navy}Quantitativas (contínuas)}

\end{itemize}

\end{itemize}
